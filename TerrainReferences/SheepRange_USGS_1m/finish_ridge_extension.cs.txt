var plan=Newtonsoft.Json.Linq.JObject.Parse(System.IO.File.ReadAllText(dir+"ridge_extension.json"));

var group=scene.GetRootGameObjects().Single(x=>x.name=="In-scene Objects").GetComponentsInChildren<UnityEngine.Transform>(true).Single(x=>x.name=="索道群");

var allPoints=new System.Collections.Generic.List<UnityEngine.Vector3[]>();
float minClearance=float.PositiveInfinity;int sampleCount=0;
for(int wire=0;wire<2;wire++){
    var cable=group.Cast<UnityEngine.Transform>().Single(t=>t.name=="Cable route "+(wire+1));
    var mesh=UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.Mesh>("Assets/TerrainData/SheepRangeDEM/Cable_"+wire+".asset");
    var localOffset=Vec(plan["cable_local_offsets"][wire]);
    var heads=towers.Select(t=>{var p=t.position+t.rotation*localOffset;p.y=t.GetComponentsInChildren<UnityEngine.Renderer>().Max(r=>r.bounds.max.y)-.7f;return p;}).ToArray();
    var path=new System.Collections.Generic.List<UnityEngine.Vector3>();
    for(int span=0;span<heads.Length-1;span++)for(int j=0;j<=(span==heads.Length-2?40:39);j++){
        float t=j/40f;var p=UnityEngine.Vector3.Lerp(heads[span],heads[span+1],t);p.y-=3*4*t*(1-t);path.Add(p);
    }
    // Sample the same piecewise parabola more densely for terrain clearance.
    for(int span=0;span<heads.Length-1;span++)for(int j=0;j<=240;j++){
        float t=j/240f;var p=UnityEngine.Vector3.Lerp(heads[span],heads[span+1],t);p.y-=3*4*t*(1-t);
        minClearance=UnityEngine.Mathf.Min(minClearance,p.y-Ground(p));sampleCount++;
    }
    var vertices=new UnityEngine.Vector3[path.Count*6];var triangles=new int[(path.Count-1)*6*6];
    for(int i=0;i<path.Count;i++){
        var forward=(path[System.Math.Min(i+1,path.Count-1)]-path[System.Math.Max(i-1,0)]).normalized;
        var right=UnityEngine.Vector3.Cross(forward,UnityEngine.Vector3.up).normalized;var up=UnityEngine.Vector3.Cross(right,forward).normalized;
        for(int j=0;j<6;j++){float angle=j*UnityEngine.Mathf.PI/3;vertices[i*6+j]=cable.InverseTransformPoint(path[i]+.10f*(right*UnityEngine.Mathf.Cos(angle)+up*UnityEngine.Mathf.Sin(angle)));}
    }
    int k=0;for(int i=0;i<path.Count-1;i++)for(int j=0;j<6;j++){int a=i*6+j,b=i*6+(j+1)%6,c=a+6,d=b+6;triangles[k++]=a;triangles[k++]=c;triangles[k++]=b;triangles[k++]=b;triangles[k++]=c;triangles[k++]=d;}

}

    var lk=locks.FirstOrDefault(l=>l.guiName==name);
    if(lk==null){lk=new MapMagic.Locks.Lock();locks.Add(lk);}

return new{waiting=false,minimumCableClearance=minClearance,samples=sampleCount,enabledLocks=mm.locks.Count(l=>l.locked)};
